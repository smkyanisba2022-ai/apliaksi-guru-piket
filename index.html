// GANTI DENGAN URL WEB APP GOOGLE APPS SCRIPT ANDA const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbyVLiApj7i0pE54XzG_oeO8u5rSNbBeem6fu_17wSUk5MWo8wLF2Haa7cjDtVqM9FEZ/exec"; let masterSiswa = []; let masterGuru = []; // 1. Inisialisasi saat aplikasi pertama kali dibuka document.addEventListener("DOMContentLoaded", () => { loadMasterData(); updateOfflineUI(); }); // 2. Fungsi Mengambil Master Data async function loadMasterData() { try { const response = await fetch(`${SCRIPT_URL}?action=get_master`); const data = await response.json(); if (data.status === "success") { masterSiswa = data.siswa; masterGuru = data.guru; // Simpan backup master data ke localStorage agar bisa dipakai saat offline localStorage.setItem("master_siswa_cache", JSON.stringify(masterSiswa)); localStorage.setItem("master_guru_cache", JSON.stringify(masterGuru)); console.log("Master Data Berhasil Dimuat dari Server"); populateDropdowns(); } else { throw new Error(data.message); } } catch (error) { console.warn("Gagal mengambil Master Data dari server. Menggunakan data cache offline...", error); masterSiswa = JSON.parse(localStorage.getItem("master_siswa_cache")) || []; masterGuru = JSON.parse(localStorage.getItem("master_guru_cache")) || []; populateDropdowns(); } } // 3. Menampilkan Master Data ke Elemen Select / Dropdown Form function populateDropdowns() { const selectSiswa = document.getElementById("selectSiswa"); const selectGuru = document.getElementById("selectGuru"); if (selectSiswa && masterSiswa.length > 0) { selectSiswa.innerHTML = ''; masterSiswa.forEach(s => { selectSiswa.innerHTML += ``; }); } if (selectGuru && masterGuru.length > 0) { selectGuru.innerHTML = ''; masterGuru.forEach(g => { selectGuru.innerHTML += ``; }); } } // 4. Fungsi Simpan Laporan Siswa (Online / Offline Fallback) async function simpanLaporanSiswa(dataLaporan) { dataLaporan.tipe = "siswa"; if (navigator.onLine) { try { const response = await fetch(SCRIPT_URL, { method: "POST", body: JSON.stringify({ action: "simpan_laporan_siswa", ...dataLaporan }) }); const res = await response.json(); if (res.status === "success") { alert("Laporan Siswa Berhasil Disimpan!"); return; } } catch (e) { console.log("Gagal koneksi, menyimpan secara offline..."); } } // Jika offline atau fetch gagal saveOffline(dataLaporan); alert("Koneksi tidak stabil. Laporan disimpan di HP secara offline."); } // 5. Fungsi Simpan Laporan Guru (Online / Offline Fallback) async function simpanLaporanGuru(dataLaporan) { dataLaporan.tipe = "guru"; if (navigator.onLine) { try { const response = await fetch(SCRIPT_URL, { method: "POST", body: JSON.stringify({ action: "simpan_laporan_guru", ...dataLaporan }) }); const res = await response.json(); if (res.status === "success") { alert("Laporan Guru Berhasil Disimpan!"); return; } } catch (e) { console.log("Gagal koneksi, menyimpan secara offline..."); } } // Jika offline atau fetch gagal saveOffline(dataLaporan); alert("Koneksi tidak stabil. Laporan disimpan di HP secara offline."); } // 6. Menyimpan Data ke Storage Offline HP function saveOffline(data) { let offlineData = JSON.parse(localStorage.getItem("laporan_offline")) || []; offlineData.push(data); localStorage.setItem("laporan_offline", JSON.stringify(offlineData)); updateOfflineUI(); } // 7. Update Indikator/Banner "X Laporan tersimpan di HP" function updateOfflineUI() { let offlineData = JSON.parse(localStorage.getItem("laporan_offline")) || []; const banner = document.getElementById("offlineBanner"); const textInfo = document.getElementById("offlineCountText"); if (banner) { if (offlineData.length > 0) { banner.style.display = "flex"; if (textInfo) textInfo.innerText = `${offlineData.length} Laporan tersimpan di HP.`; } else { banner.style.display = "none"; } } } // 8. FUNGSI UTAMA: Upload Semua Data Offline Sekaligus (Fix Bug Hanya 1 Data) async function uploadDataOffline() { let offlineData = JSON.parse(localStorage.getItem("laporan_offline")) || []; if (offlineData.length === 0) { alert("Tidak ada data offline untuk diupload."); return; } // Memisahkan data siswa dan guru let listSiswa = offlineData.filter(item => item.tipe === "siswa"); let listGuru = offlineData.filter(item => item.tipe === "guru"); try { // Send Batch Siswa if (listSiswa.length > 0) { const resSiswa = await fetch(SCRIPT_URL, { method: "POST", body: JSON.stringify({ action: "simpan_batch_siswa", items: listSiswa }) }); const resultSiswa = await resSiswa.json(); if (resultSiswa.status !== "success") throw new Error(resultSiswa.message); } // Send Batch Guru if (listGuru.length > 0) { const resGuru = await fetch(SCRIPT_URL, { method: "POST", body: JSON.stringify({ action: "simpan_batch_guru", items: listGuru }) }); const resultGuru = await resGuru.json(); if (resultGuru.status !== "success") throw new Error(resultGuru.message); } alert(`Berhasil mengunggah ${offlineData.length} laporan ke Google Spreadsheet!`); // Hapus penyimpanan lokal HANYA jika semua berhasil terunggah localStorage.removeItem("laporan_offline"); updateOfflineUI(); } catch (error) { console.error("Gagal melakukan batch upload:", error); alert("Gagal mengupload data. Pastikan koneksi internet stabil lalu coba lagi."); } }
